WITH monthly AS (
SELECT
s.name AS store,
DATE_TRUNC('month', o.ordered_at) AS month,
SUM(o.order_total) AS revenue
FROM orders o
JOIN stores s ON s.id = o.store_id
GROUP BY s.name, DATE_TRUNC('month', o.ordered_at)
)
SELECT
store,
month,
revenue,
(revenue - LAG(revenue) OVER (PARTITION BY store ORDER BY month))
/ NULLIF(LAG(revenue) OVER (PARTITION BY store ORDER BY month), 0) AS mom_pct_change
FROM monthly
ORDER BY store, month;
